# TODO: check this
CUR_TTY=/dev/stdout

if [ -f "/app/bin/retrodeck.sh" ]; then
  # loading the RetroDECK framework that even give access to variables such as roms_folder
  if [ -z "$roms_folder" ]; then
    source /app/libexec/global.sh
  fi

  if [ -z "$rdhome" ] && [ -z "$rd_home_path" ]; then
    echo "Something went wrong."
    exit 255
  fi

  if [ -z "$ports_folder" ]; then
    if [ ! -z "$rd_home_path" ]; then
      ports_folder="$rd_home_path/PortMaster"
    else
      ports_folder="$rdhome/PortMaster"
    fi
  fi

  export controlfolder="/var/data/PortMaster"
  export directory="$ports_folder"
elif [ -f "/var/config/retrodeck/retrodeck.json" ]; then
  # Fallback
  ports_folder="$(jq -r ".paths.ports_path" "/var/config/retrodeck/retrodeck.json")"

  if [ "$ports_folder" = "null" ]; then
    ports_folder="$(jq -r ".paths.rd_home_path" "/var/config/retrodeck/retrodeck.json")/PortMaster"
  fi

  export controlfolder="/var/data/PortMaster"
  export directory="$ports_folder"
elif [ -f ~/.var/app/net.retrodeck.retrodeck/config/retrodeck/retrodeck.json ]; then
  # Fallback
  ports_folder="$(jq -r ".paths.ports_path" ~/.var/app/net.retrodeck.retrodeck/config/retrodeck/retrodeck.json)"

  if [ "$ports_folder" = "null" ]; then
    ports_folder="$(jq -r ".paths.rd_home_path" ~/.var/app/net.retrodeck.retrodeck/config/retrodeck/retrodeck.json)/PortMaster"
  fi

  export controlfolder="$HOME/.var/app/net.retrodeck.retrodeck/data/PortMaster"
  export directory="$ports_folder"
elif [ -f "/var/config/retrodeck/retrodeck.cfg" ]; then
  # Fallback
  ports_folder="$(grep "ports_folder" /var/config/retrodeck/retrodeck.cfg | awk -F= '{print $2}')"

  if [ -z "$ports_folder" ]; then
    rdhome="$(grep "rdhome" /var/config/retrodeck/retrodeck.cfg | awk -F= '{print $2}')"

    ports_folder="$rdhome/PortMaster"
  fi

  export controlfolder="/var/data/PortMaster"
  export directory="$ports_folder"
else
  # Another Fallback
  ports_folder="$(grep "ports_folder" ~/.var/app/net.retrodeck.retrodeck/config/retrodeck/retrodeck.cfg | awk -F= '{print $2}')"

  if [ -z "$ports_folder" ]; then
    rdhome="$(grep "rdhome" ~/.var/app/net.retrodeck.retrodeck/config/retrodeck/retrodeck.cfg | awk -F= '{print $2}')"

    ports_folder="$rdhome/PortMaster"
  fi

  export controlfolder="$HOME/.var/app/net.retrodeck.retrodeck/data/PortMaster"
  export directory="$ports_folder"
fi

echo "\$controlfolder = $controlfolder"
echo "\$directory = $directory"

# Select the bundled tools and libraries for the host architecture.
case "$(uname -m 2>/dev/null)" in
  x86_64|amd64) RETRODECK_ARCH="x86_64" ;;
  aarch64|arm64) RETRODECK_ARCH="aarch64" ;;
  armv7*|armhf|armv8l) RETRODECK_ARCH="armhf" ;;
  *)
    echo "Unsupported RetroDECK architecture: $(uname -m 2>/dev/null)" >&2
    exit 255
    ;;
esac

RETRODECK_ARCH_DIR="$controlfolder/retrodeck/$RETRODECK_ARCH"
if [ ! -d "$RETRODECK_ARCH_DIR" ]; then
  echo "No RetroDECK tools are installed for $RETRODECK_ARCH." >&2
  exit 255
fi

# Unpacking PortMaster.zip doesn't always keep the executable bit, and the checks below need it before
# harbourmaster ever runs (its RetroDECK platform hook fixes the same files only once pugwash is up).
for tool in gptokeyb gptokeyb2 xdelta3 unsquashfs; do
  if [ -f "$RETRODECK_ARCH_DIR/$tool" ] && [ ! -x "$RETRODECK_ARCH_DIR/$tool" ]; then
    chmod +x "$RETRODECK_ARCH_DIR/$tool"
  fi
done

# Use architecture-matched tools. PortMaster updates can replace the unsuffixed
# root helpers, so RetroDECK keeps its AArch64 copies in its private tool dir.
RETRODECK_TOOL_BIN="$controlfolder/retrodeck/.arch-tools/$RETRODECK_ARCH"
mkdir -p "$RETRODECK_TOOL_BIN"
case "$RETRODECK_ARCH" in
  x86_64)
    TOOL_SUFFIX=".x86_64"
    TOOL_SOURCE_DIR="$controlfolder"
    ;;
  armhf)
    TOOL_SUFFIX=".armhf"
    TOOL_SOURCE_DIR="$controlfolder"
    ;;
  aarch64)
    TOOL_SUFFIX=""
    TOOL_SOURCE_DIR="$RETRODECK_ARCH_DIR"
    ;;
esac
for tool in gptokeyb gptokeyb2 xdelta3; do
  if [ -x "$TOOL_SOURCE_DIR/$tool$TOOL_SUFFIX" ]; then
    ln -sfn "$TOOL_SOURCE_DIR/$tool$TOOL_SUFFIX" "$RETRODECK_TOOL_BIN/$tool"
  elif [ "$RETRODECK_ARCH" = "aarch64" ]; then
    rm -f "$RETRODECK_TOOL_BIN/$tool"
  fi
done

if [ "$RETRODECK_ARCH" = "aarch64" ] && { [ ! -x "$RETRODECK_ARCH_DIR/gptokeyb" ] || [ ! -x "$RETRODECK_ARCH_DIR/gptokeyb2" ]; }; then
  echo "AArch64 gptokeyb and gptokeyb2 must be installed in $RETRODECK_ARCH_DIR; refusing to use the x86_64 PortMaster binaries." >&2
  exit 255
fi

export LD_LIBRARY_PATH="/usr/lib:/lib:/app/retrodeck/components/shared-libs:/app/retrodeck/components/shared-libs/org.kde.Sdk/6.7/:/app/retrodeck/components/shared-libs:/app/retrodeck/components/shared-libs/org.kde.Platform/6.7/:/app/retrodeck/components/shared-libs/org.gnome.Platform/49/:/app/retrodeck/components/shared-libs/org.freedesktop.Platform/25.08/:$LD_LIBRARY_PATH"
export PYSDL2_DLL_PATH="$LD_LIBRARY_PATH"
export ESUDO=""
export ESUDOKILL="-k" # for 351Elec and EmuELEC use "-1" (numeric one) or "-k" 

# TODO: might need to add other devices here.
DEVICE="03000000de280000ff11000001000000"

export SDL_GAMECONTROLLERCONFIG_FILE="$controlfolder/gamecontrollerdb.txt"
export sdl_controllerconfig="$(grep "$SDL_GAMECONTROLLERCONFIG_FILE" -e "${DEVICE}")"

echo "\$sdl_controllerconfig = $sdl_controllerconfig"

# Enable mount hack :D
export PATH="$RETRODECK_TOOL_BIN:$RETRODECK_ARCH_DIR:$controlfolder/retrodeck:$PATH"
export LD_LIBRARY_PATH="$RETRODECK_ARCH_DIR:$controlfolder/retrodeck/libs.$RETRODECK_ARCH:$controlfolder/retrodeck:$LD_LIBRARY_PATH"

# Opt-in GLES1 compatibility runner for individual ports.
if [ -f "$controlfolder/retrodeck/gles1_compat.sh" ]; then
  . "$controlfolder/retrodeck/gles1_compat.sh"
fi

get_controls() {
  ANALOGSTICKS="2"
  LOWRES="N"
}

export PM_CAN_MOUNT="N"
source $controlfolder/device_info.txt
source $controlfolder/funcs.txt

if [ "$RETRODECK_ARCH" = "aarch64" ]; then
  export GPTOKEYB2="$ESUDO env LD_PRELOAD=$controlfolder/libinterpose.${DEVICE_ARCH}.so $RETRODECK_ARCH_DIR/gptokeyb2 $ESUDOKILL"
  export GPTOKEYB="$ESUDO $RETRODECK_ARCH_DIR/gptokeyb $ESUDOKILL"
else
  export GPTOKEYB2="$ESUDO env LD_PRELOAD=$controlfolder/libinterpose.${DEVICE_ARCH}.so $controlfolder/gptokeyb2 $ESUDOKILL"
  export GPTOKEYB="$ESUDO $controlfolder/gptokeyb $ESUDOKILL"
fi
